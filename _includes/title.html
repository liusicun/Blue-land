{%- comment -%} Prints a title so its last word (or last Chinese character) and the blue dot never get split onto separate lines. Use: {% include title.html t=post.title dot=... %} {%- endcomment -%} {%- assign _t = include.t | strip -%} {%- assign _w = _t | split: ' ' -%} {%- if _w.size > 1 -%} {%- assign _last = _w | last -%}{%- assign _head = _w | pop | join: ' ' -%} {{ _head }} {{ _last }}{{ include.dot }} {%- else -%} {%- assign _n = _t | size | minus: 1 -%} {%- if _n > 0 -%}{{ _t | slice: 0, _n }}{{ _t | slice: _n, 1 }}{{ include.dot }}{%- else -%}{{ _t }}{{ include.dot }}{%- endif -%} {%- endif -%}
